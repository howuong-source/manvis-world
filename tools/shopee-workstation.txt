<script>
    function calculatePrices() {
        const target = parseFloat(document.getElementById('targetProfit').value);
        
        if (isNaN(target) || target <= 0) {
            document.getElementById('shopeeNormalPrice').innerText = '請輸入有效金額';
            document.getElementById('shopeePromoPrice').innerText = '請輸入有效金額';
            document.getElementById('familyPrice').innerText = '請輸入有效金額';
            return;
        }

        const invoiceFee = 2.5; 

        // 1. 蝦皮平常日
        const shopeeNormalRaw = (target + invoiceFee) / 0.86;
        const shopeeNormalFinal = Math.ceil(shopeeNormalRaw); 

        // 2. 蝦皮大促日
        const shopeePromoRaw = (target + invoiceFee) / 0.84;
        const shopeePromoFinal = Math.ceil(shopeePromoRaw); 

        // 3. 全家好賣+
        const familyFinal = target;

        document.getElementById('shopeeNormalPrice').innerText = `改賣 ${shopeeNormalFinal} 元`;
        document.getElementById('shopeeNormalFormula').innerText = `精確算式：(${target} + 2.5) ÷ 0.86 = ${shopeeNormalRaw.toFixed(2)} 元 (採無條件進位)`;

        document.getElementById('shopeePromoPrice').innerText = `改賣 ${shopeePromoFinal} 元`;
        document.getElementById('shopeePromoFormula').innerText = `精確算式：(${target} + 2.5) ÷ 0.84 = ${shopeePromoRaw.toFixed(2)} 元 (採無條件進位)`;

        document.getElementById('familyPrice').innerText = `定價 ${familyFinal} 元`;
        document.getElementById('familyFormula').innerText = `精確算式：直接設定 ${familyFinal} 元（不抽成，買家自付 35 元運費）`;
    }

    calculatePrices();
</script>
